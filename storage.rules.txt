// Firebase 콘솔 > Storage > 규칙(Rules) 에 아래 내용을 통째로 붙여넣고 [게시]를 누르세요.
rules_version = '2';
service firebase.storage {
  match /b/{bucket}/o {
    match /u/{code}/{fileName} {
      // 코드를 아는 사람만 그 코드의 사진을 보고 지울 수 있습니다.
      allow read, delete: if true;
      // 사진 파일만, 한 장에 30MB까지 올릴 수 있습니다.
      allow create: if request.resource.size > 0
                    && request.resource.size < 30 * 1024 * 1024
                    && request.resource.contentType.matches('image/.*');
    }
  }
}
